(function () {
  const DATA = window.MODEL_DATA;
  const params = new URLSearchParams(location.search);
  const view = params.get("view") || "interactive";

  const HUMAN = "#1d4e89";
  const COPILOT = "#b35a16";
  const PIPELINE = "#1b6a49";
  const CURRENT = "#8a8578";
  const INK = "#1c1b18";
  const MUTED = "#5a564c";

  function color(actor, mode) {
    if (mode === "current") return CURRENT;
    if (actor === "Human") return HUMAN;
    if (actor === "Copilot") return COPILOT;
    return PIPELINE;
  }

  function wrap(txt, max) {
    const words = String(txt).split(" ");
    const lines = [];
    let cur = "";
    words.forEach((w) => {
      if ((cur + " " + w).trim().length > max) { lines.push(cur.trim()); cur = w; }
      else cur = (cur + " " + w).trim();
    });
    if (cur) lines.push(cur);
    return lines;
  }

  function escapeXml(s) {
    return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
  }
  function escapeHtml(s) {
    return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
  }

  function pdsaSvg(step, mode, opts) {
    opts = opts || {};
    const W = opts.W || 600, H = opts.H || 340;
    const cx = opts.cx || 300, cy = opts.cy || 170, r = opts.r || 92;
    const sw = opts.sw || 14;
    const fs = opts.fs || 14;
    const fl = opts.fl || 12;
    const max = opts.max || 26;
    const quads = mode === "future"
      ? [
          { k: "Plan", q: step.fut.plan },
          { k: "Do", q: step.fut.doing },
          { k: "Study", q: step.fut.study },
          { k: "Act", q: step.fut.act }
        ]
      : [
          { k: "Plan", q: { text: step.cur.plan, actor: "Human" } },
          { k: "Do", q: { text: step.cur.doing, actor: "Human" } },
          { k: "Study", q: { text: step.cur.study, actor: "Human" } },
          { k: "Act", q: { text: step.cur.act, actor: "Human" } }
        ];
    function arc(i) {
      const a0 = (-90 + i * 90 + 4) * Math.PI / 180;
      const a1 = (-90 + (i + 1) * 90 - 4) * Math.PI / 180;
      return "M " + (cx + r * Math.cos(a0)) + " " + (cy + r * Math.sin(a0)) +
        " A " + r + " " + r + " 0 0 1 " + (cx + r * Math.cos(a1)) + " " + (cy + r * Math.sin(a1));
    }
    const pos = opts.pos || [
      { x: 410, y: 58, anchor: "start" },
      { x: 410, y: 268, anchor: "start" },
      { x: 190, y: 268, anchor: "end" },
      { x: 190, y: 58, anchor: "end" }
    ];
    let paths = "";
    quads.forEach((x, i) => {
      paths += '<path d="' + arc(i) + '" fill="none" stroke="' + color(x.q.actor, mode) +
        '" stroke-width="' + sw + '" stroke-linecap="round"></path>';
    });
    const mid1 = mode === "future" ? "Repeat" : "Slow loop";
    const mid2 = mode === "future" ? "until green" : "study comes late";
    let texts = '<text x="' + cx + '" y="' + (cy - 6) + '" text-anchor="middle" fill="' + INK +
      '" font-size="' + (fs + 2) + '" font-weight="650">' + mid1 + "</text>" +
      '<text x="' + cx + '" y="' + (cy + 16) + '" text-anchor="middle" fill="' + MUTED +
      '" font-size="' + fl + '">' + mid2 + "</text>";
    quads.forEach((x, i) => {
      const p = pos[i];
      const label = x.k + (mode === "future" ? " · " + x.q.actor : "");
      const lines = wrap(x.q.text, max);
      texts += '<text x="' + p.x + '" y="' + p.y + '" text-anchor="' + p.anchor + '" fill="' + INK +
        '" font-size="' + fs + '" font-weight="650">' + escapeXml(label) + "</text>";
      lines.forEach((ln, j) => {
        texts += '<text x="' + p.x + '" y="' + (p.y + 20 + j * (fl + 6)) + '" text-anchor="' + p.anchor +
          '" fill="' + MUTED + '" font-size="' + fl + '">' + escapeXml(ln) + "</text>";
      });
    });
    return '<svg class="pdsa-svg" viewBox="0 0 ' + W + " " + H +
      '" width="100%" xmlns="http://www.w3.org/2000/svg" role="img" aria-label="PDSA loop">' +
      paths + texts + "</svg>";
  }

  function contractGraph(sel, opts) {
    opts = opts || {};
    const W = opts.W || 1000, H = opts.H || 540;
    const cx = W / 2, cy = H / 2;
    const rx = opts.rx || 380, ry = opts.ry || 205;
    const w = opts.w || 170, h = opts.h || 40;
    const fs = opts.fs || 13;
    const n = DATA.contracts.length;
    const nodes = DATA.contracts.map((c, i) => {
      const a = (-90 + i * (360 / n)) * Math.PI / 180;
      return { c: c, x: cx + rx * Math.cos(a), y: cy + ry * Math.sin(a) };
    });
    let s = "";
    nodes.forEach((node, i) => {
      const on = i === sel;
      s += '<line x1="' + cx + '" y1="' + cy + '" x2="' + node.x + '" y2="' + node.y +
        '" stroke="' + (on ? HUMAN : "#c8c1b3") + '" stroke-width="' + (on ? 3 : 1.25) + '"></line>';
    });
    const cw = opts.cw || 220, ch = opts.ch || 64;
    s += '<rect x="' + (cx - cw / 2) + '" y="' + (cy - ch / 2) + '" width="' + cw + '" height="' + ch +
      '" rx="12" fill="#fffcf6" stroke="#1c1b18" stroke-width="1.5"></rect>';
    s += '<text x="' + cx + '" y="' + (cy - 4) + '" text-anchor="middle" fill="' + INK +
      '" font-size="' + (fs + 3) + '" font-weight="650">Checkout team</text>';
    s += '<text x="' + cx + '" y="' + (cy + 16) + '" text-anchor="middle" fill="' + MUTED +
      '" font-size="' + fs + '">about 50 people</text>';
    nodes.forEach((node, i) => {
      const on = i === sel;
      s += '<g data-node="' + i + '" style="cursor:pointer">';
      s += '<rect x="' + (node.x - w / 2) + '" y="' + (node.y - h / 2) + '" width="' + w + '" height="' + h +
        '" rx="10" fill="' + (on ? "#e7eef7" : "#fffcf6") + '" stroke="' + (on ? HUMAN : "#c8c1b3") +
        '" stroke-width="' + (on ? 2 : 1) + '"></rect>';
      s += '<text x="' + node.x + '" y="' + (node.y + 5) + '" text-anchor="middle" fill="' + INK +
        '" font-size="' + fs + '" font-weight="' + (on ? 650 : 500) + '">' + escapeXml(node.c.who) + "</text>";
      s += "</g>";
    });
    return '<svg class="graph-svg" viewBox="0 0 ' + W + " " + H +
      '" width="100%" xmlns="http://www.w3.org/2000/svg" role="img" aria-label="Contract graph">' +
      s + "</svg>";
  }

  const bigOpts = {
    W: 700, H: 420, cx: 350, cy: 210, r: 118, sw: 22, fs: 18, fl: 16, max: 28,
    pos: [
      { x: 490, y: 70, anchor: "start" },
      { x: 490, y: 330, anchor: "start" },
      { x: 210, y: 330, anchor: "end" },
      { x: 210, y: 70, anchor: "end" }
    ]
  };

  function mountAllPdsa() {
    document.querySelectorAll(".pdsa-mount").forEach((el) => {
      const i = Number(el.getAttribute("data-step"));
      const mode = el.getAttribute("data-mode");
      const big = el.getAttribute("data-big") === "1";
      el.innerHTML = pdsaSvg(DATA.steps[i], mode, big ? bigOpts : {});
    });
  }

  function formatGate(text) {
    const items = String(text).split("\n").map((line) => {
      line = line.trim();
      if (!line) return "";
      let label = line.split(/\s+/)[0];
      let rest = line.slice(label.length).trim();
      if (label === "NOT") { label = "NOT A CHECK"; rest = rest.replace(/^A CHECK\s*/, ""); }
      if (label === "LANDS") { label = "LANDS IN"; rest = rest.replace(/^IN\s*/, ""); }
      let kind = "green";
      if (label.indexOf("NOT") === 0) kind = "not";
      else if (["ACCEPTED", "DERIVED", "LANDS IN"].indexOf(label) >= 0) kind = "info";
      return '<li class="gate-' + kind + '"><span class="gate-label">' + escapeHtml(label) + "</span> " + escapeHtml(rest) + "</li>";
    }).join("");
    return '<h5>Gate</h5><ul class="gate">' + items + "</ul>";
  }

  let sel = 2;
  function setMode(mode) {
    document.body.setAttribute("data-mode", mode);
    document.querySelectorAll(".mode-btn").forEach((b) => {
      b.setAttribute("aria-pressed", b.getAttribute("data-mode") === mode ? "true" : "false");
    });
    const edges = document.getElementById("stream-edges");
    if (edges) {
      edges.innerHTML = mode === "future"
        ? "<span>Correction edge (short): Verify or Review returns one unit to Build</span> · <span>Learning edge (long): Operate and learn sends a constraint back to Intake</span>"
        : "Return path: defects found late go back as a batch to the next release";
    }
    const legend = document.getElementById("legend");
    if (legend) legend.hidden = mode !== "future";
    renderZoom();
  }

  function renderZoom() {
    const mode = document.body.getAttribute("data-mode") || "future";
    const step = DATA.steps[sel];
    document.querySelectorAll(".stream-card").forEach((b) => {
      b.setAttribute("aria-pressed", Number(b.getAttribute("data-step")) === sel ? "true" : "false");
    });
    const title = document.getElementById("zoom-title");
    if (title) title.textContent = "PDSA loop inside step " + (sel + 1) + ": " + step.name;
    const mount = document.getElementById("zoom-pdsa");
    if (mount) mount.innerHTML = pdsaSvg(step, mode);
    const side = document.getElementById("zoom-side");
    if (!side) return;
    const stitle = document.getElementById("zoom-side-title");
    if (stitle) stitle.textContent = mode === "future" ? "CD future state" : "Current state";
    if (mode === "future") {
      side.innerHTML =
        '<dl class="props">' +
        "<div><dt>What happens</dt><dd>" + escapeHtml(step.fut.summary) + "</dd></div>" +
        "<div><dt>Copilot</dt><dd>" + escapeHtml(step.fut.copilot) + "</dd></div>" +
        "<div><dt>Human</dt><dd>" + escapeHtml(step.fut.human) + "</dd></div>" +
        "<div><dt>Lane</dt><dd>" + escapeHtml(step.fut.lane) + "</dd></div>" +
        '<div><dt>Time</dt><dd><span class="blank">fill in</span></dd></div>' +
        "</dl>" + formatGate(step.fut.gate);
    } else {
      side.innerHTML =
        '<dl class="props">' +
        "<div><dt>What happens</dt><dd>" + escapeHtml(step.cur.summary) + "</dd></div>" +
        '<div><dt>Wait / cycle time</dt><dd><span class="blank">fill in</span></dd></div>' +
        "<div><dt>Article role</dt><dd>" + escapeHtml(step.role) + "</dd></div>" +
        "</dl>";
    }
  }

  let csel = 1;
  function renderGraph(host, selected, opts) {
    if (!host) return;
    host.innerHTML = contractGraph(selected, opts);
    host.querySelectorAll("[data-node]").forEach((g) => {
      g.addEventListener("click", function () {
        selectContract(Number(g.getAttribute("data-node")));
      });
    });
  }

  function selectContract(i) {
    csel = i;
    const selEl = document.getElementById("contract-select");
    if (selEl) selEl.value = String(i);
    renderGraph(document.getElementById("contract-graph"), csel);
    document.querySelectorAll(".interactive-contracts .contract-card").forEach((card) => {
      card.hidden = Number(card.getAttribute("data-contract")) !== i;
    });
  }

  function setExpanded(on) {
    document.body.classList.toggle("expanded", on);
    document.querySelectorAll(".expand-btn").forEach((b) => {
      b.setAttribute("aria-pressed", on ? "true" : "false");
    });
  }

  if (view === "print") {
    document.body.classList.add("expanded");
    document.documentElement.classList.add("print-ready");
  } else if (["stream", "pdsa", "contracts", "roles", "team"].indexOf(view) >= 0) {
    document.body.classList.add("slide-view", "view-" + view);
  }

  mountAllPdsa();
  if (view === "interactive" || view === "print") {
    setMode("future");
    document.querySelectorAll(".mode-btn").forEach((b) => {
      b.addEventListener("click", function () { setMode(b.getAttribute("data-mode")); });
    });
    document.querySelectorAll(".stream-card").forEach((b) => {
      b.addEventListener("click", function () {
        sel = Number(b.getAttribute("data-step"));
        renderZoom();
      });
    });
    document.querySelectorAll(".expand-btn").forEach((b) => {
      b.addEventListener("click", function () {
        setExpanded(!document.body.classList.contains("expanded"));
      });
    });
    const printBtn = document.getElementById("print-btn");
    if (printBtn) printBtn.addEventListener("click", function () { window.print(); });
    const cselEl = document.getElementById("contract-select");
    if (cselEl) cselEl.addEventListener("change", function (e) { selectContract(Number(e.target.value)); });
    renderGraph(document.getElementById("contract-graph"), csel);
    renderGraph(document.getElementById("contract-graph-print"), -1, { W: 1000, H: 540 });
  }
  if (view === "contracts") {
    renderGraph(document.getElementById("slide-graph"), -1, {
      W: 1480, H: 620, rx: 560, ry: 230, w: 230, h: 52, fs: 18, cw: 280, ch: 80
    });
  }
  if (view === "pdsa") {
    const el = document.getElementById("slide-pdsa-mount");
    if (el) el.innerHTML = pdsaSvg(DATA.steps[2], "future", bigOpts);
  }
})();

</script>
</body>
</html>
